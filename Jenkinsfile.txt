pipeline {
    agent any
    environment {
        APP_NAME = 'devops-runbook'
        ENV_NAME = 'development'
    }
    stages {
        stage('Build') {
            steps {
                echo "Building ${APP_NAME}..."
            }
        }
        stage('Test') {
            steps {
                echo "Testing ${APP_NAME}..."
            }
        }
        stage('Deploy') {
            steps {
                echo "Deploying ${APP_NAME}..."
            }
        }
    }
    post {
        success {
            echo 'Pipeline SUCCEEDED!'
        }
        failure {
            echo 'Pipeline FAILED!'
        }
    }
}
